    <!-- Aplicação Lógica -->
    <script>
        const GROUPS_DEF = [
            "Motor Diesel",
            "Sistema Hidráulico",
            "Chassi",
            "Sistema de Trabalho",
            "Cabine",
            "Sistema Elétrico"
        ];

        class FleetCheckApp {
            constructor() {
                this.inspections = JSON.parse(localStorage.getItem('xcmg_inspections') || '[]');
                this.currentInspectionId = null;
                this.formData = this.getEmptyForm();
                this.init();
            }

            getEmptyForm() {
                const data = {
                    id: 'XCMG-' + Date.now(),
                    date: new Date().toISOString().split('T')[0],
                    model: '',
                    serial: '',
                    meter: '',
                    location: '',
                    inspector: '',
                    eqPhoto: null,
                    serialPhoto: null,
                    groups: {}
                };
                GROUPS_DEF.forEach(group => {
                    data.groups[group] = {
                        status: 'OK',
                        obs: '',
                        photo: null,
                        parts: []
                    };
                });
                return data;
            }

            init() {
                this.renderGroupsForm();
                document.getElementById('eq-date').value = new Date().toISOString().split('T')[0];
            }

            renderGroupsForm() {
                const container = document.getElementById('groups-container');
                container.innerHTML = '';

                GROUPS_DEF.forEach((groupName, index) => {
                    const gData = this.formData.groups[groupName];
                    const div = document.createElement('div');
                    div.className = 'bg-xcmgDark border border-gray-800 rounded-xl p-5 shadow-md space-y-4';
                    
                    div.innerHTML = `
                        <div class="flex flex-col sm:flex-row justify-between items-start sm:items-center border-b border-gray-800 pb-3 gap-2">
                            <h3 class="font-bold text-white flex items-center text-base">
                                <span class="bg-gray-800 text-xcmgYellow text-xs px-2 py-1 rounded mr-2">${index + 1}</span>
                                ${groupName}
                            </h3>
                            <div class="flex space-x-2 w-full sm:w-auto">
                                <button type="button" onclick="app.setGroupStatus('${groupName}', 'OK')" class="flex-1 sm:flex-none px-3 py-1.5 rounded-lg text-xs font-bold transition ${gData.status === 'OK' ? 'bg-green-600 text-white shadow' : 'bg-gray-800 text-gray-400 hover:bg-gray-700'}">OK</button>
                                <button type="button" onclick="app.setGroupStatus('${groupName}', 'Atencao')" class="flex-1 sm:flex-none px-3 py-1.5 rounded-lg text-xs font-bold transition ${gData.status === 'Atencao' ? 'bg-yellow-600 text-white shadow' : 'bg-gray-800 text-gray-400 hover:bg-gray-700'}">Atenção</button>
                                <button type="button" onclick="app.setGroupStatus('${groupName}', 'Critico')" class="flex-1 sm:flex-none px-3 py-1.5 rounded-lg text-xs font-bold transition ${gData.status === 'Critico' ? 'bg-red-600 text-white shadow' : 'bg-gray-800 text-gray-400 hover:bg-gray-700'}">Crítico</button>
                            </div>
                        </div>

                        <!-- Foto e Observações do Grupo -->
                        <div class="grid grid-cols-1 md:grid-cols-3 gap-4">
                            <div class="md:col-span-2 space-y-3">
                                <div>
                                    <label class="block text-xs font-semibold uppercase text-gray-400 mb-1">Observações do Grupo</label>
                                    <textarea rows="2" oninput="app.updateGroupObs('${groupName}', this.value)" placeholder="Descreva falhas, vazamentos ou condições..." class="w-full bg-gray-900 border border-gray-700 rounded-lg p-2.5 text-white focus:border-xcmgYellow focus:outline-none text-sm">${gData.obs}</textarea>
                                </div>
                            </div>
                            <div class="flex flex-col items-center">
                                <label class="block text-xs font-semibold uppercase text-gray-400 mb-1 self-start">Foto do Grupo</label>
                                <div id="preview-group-${index}" class="w-full h-24 bg-gray-900 rounded-lg flex items-center justify-center border border-gray-700 mb-2 overflow-hidden relative">
                                    ${gData.photo ? `<img src="${gData.photo}" class="w-full h-full object-cover">` : `<span class="text-xs text-gray-500">Sem foto</span>`}
                                </div>
                                <div class="flex space-x-1 w-full">
                                    <label class="flex-1 bg-gray-800 hover:bg-gray-700 text-center py-1.5 rounded text-xs font-medium cursor-pointer border border-gray-700 transition">
                                        <i class="fa-solid fa-camera mr-1"></i> Câmera
                                        <input type="file" accept="image/*" capture="environment" class="hidden" onchange="app.handleGroupPhoto(event, '${groupName}')">
                                    </label>
                                    <label class="flex-1 bg-gray-800 hover:bg-gray-700 text-center py-1.5 rounded text-xs font-medium cursor-pointer border border-gray-700 transition">
                                        <i class="fa-solid fa-image mr-1"></i> Galeria
                                        <input type="file" accept="image/*" class="hidden" onchange="app.handleGroupPhoto(event, '${groupName}')">
                                    </label>
                                </div>
                            </div>
                        </div>

                        <!-- Lista de Peças Manual por Grupo -->
                        <div class="border-t border-gray-800 pt-3 space-y-3">
                            <div class="flex justify-between items-center">
                                <span class="text-xs font-bold uppercase text-gray-400"><i class="fa-solid fa-gears text-xcmgYellow mr-1"></i> Lista de Peças Necessárias</span>
                                <button type="button" onclick="app.addPartRow('${groupName}')" class="bg-gray-800 hover:bg-gray-700 text-xcmgYellow px-2.5 py-1 rounded text-xs font-semibold transition border border-gray-700">
                                    <i class="fa-solid fa-plus mr-1"></i> Adicionar Peça
                                </button>
                            </div>
                            <div id="parts-list-${index}" class="space-y-2">
                                ${gData.parts.length === 0 ? `<p class="text-xs text-gray-500 italic">Nenhuma peça adicionada manualmente.</p>` : ''}
                                ${gData.parts.map((part, pIndex) => `
                                    <div class="flex items-center space-x-2 bg-gray-900 p-2 rounded-lg border border-gray-800">
                                        <input type="text" value="${part.desc}" oninput="app.updatePart('${groupName}', ${pIndex}, 'desc', this.value)" placeholder="Descrição da Peça" class="flex-1 bg-gray-800 border border-gray-700 rounded p-1.5 text-xs text-white focus:border-xcmgYellow focus:outline-none">
                                        <input type="text" value="${part.pn}" oninput="app.updatePart('${groupName}', ${pIndex}, 'pn', this.value)" placeholder="Part Number (PN)" class="w-32 bg-gray-800 border border-gray-700 rounded p-1.5 text-xs text-white focus:border-xcmgYellow focus:outline-none uppercase">
                                        <input type="number" value="${part.qty}" oninput="app.updatePart('${groupName}', ${pIndex}, 'qty', this.value)" placeholder="Qtd" min="1" class="w-16 bg-gray-800 border border-gray-700 rounded p-1.5 text-xs text-white focus:border-xcmgYellow focus:outline-none text-center">
                                        <button type="button" onclick="app.removePartRow('${groupName}', ${pIndex})" class="text-red-400 hover:text-red-300 p-1.5"><i class="fa-solid fa-trash"></i></button>
                                    </div>
                                `).join('')}
                            </div>
                        </div>
                    `;
                    container.appendChild(div);
                });
            }

            handleInitialPhoto(event, type) {
                const file = event.target.files[0];
                if (!file) return;
                const reader = new FileReader();
                reader.onload = (e) => {
                    this.formData[type] = e.target.result;
                    const previewId = type === 'eqPhoto' ? 'preview-eq-photo' : 'preview-serial-photo';
                    document.getElementById(previewId).innerHTML = `<img src="${e.target.result}" class="w-full h-full object-cover">`;
                };
                reader.readAsDataURL(file);
            }

            handleGroupPhoto(event, groupName) {
                const file = event.target.files[0];
                if (!file) return;
                const reader = new FileReader();
                reader.onload = (e) => {
                    this.formData.groups[groupName].photo = e.target.result;
                    this.renderGroupsForm();
                };
                reader.readAsDataURL(file);
            }

            setGroupStatus(groupName, status) {
                this.formData.groups[groupName].status = status;
                this.renderGroupsForm();
            }

            updateGroupObs(groupName, val) {
                this.formData.groups[groupName].obs = val;
            }

            addPartRow(groupName) {
                this.formData.groups[groupName].parts.push({ desc: '', pn: '', qty: 1 });
                this.renderGroupsForm();
            }

            updatePart(groupName, index, field, value) {
                this.formData.groups[groupName].parts[index][field] = value;
            }

            removePartRow(groupName, index) {
                this.formData.groups[groupName].parts.splice(index, 1);
                this.renderGroupsForm();
            }

            syncFormFromInputs() {
                this.formData.model = document.getElementById('eq-model').value;
                this.formData.serial = document.getElementById('eq-serial').value;
                this.formData.meter = document.getElementById('eq-meter').value;
                this.formData.location = document.getElementById('eq-location').value;
                this.formData.inspector = document.getElementById('eq-inspector').value;
                this.formData.date = document.getElementById('eq-date').value;
            }

            saveInspection() {
                this.syncFormFromInputs();
                
                if (!this.formData.model) {
                    alert('Por favor, informe o Modelo do Equipamento.');
                    document.getElementById('eq-model').focus();
                    return;
                }

                const existingIndex = this.inspections.findIndex(i => i.id === this.formData.id);
                if (existingIndex >= 0) {
                    this.inspections[existingIndex] = this.formData;
                } else {
                    this.inspections.unshift(this.formData);
                }

                localStorage.setItem('xcmg_inspections', JSON.stringify(this.inspections));
                alert('Inspeção salva com sucesso!');
                this.showHistory();
            }

            resetForm() {
                if (confirm('Deseja limpar todos os campos da inspeção atual?')) {
                    this.currentInspectionId = null;
                    this.formData = this.getEmptyForm();
                    document.getElementById('eq-model').value = '';
                    document.getElementById('eq-serial').value = '';
                    document.getElementById('eq-meter').value = '';
                    document.getElementById('eq-location').value = '';
                    document.getElementById('eq-inspector').value = '';
                    document.getElementById('preview-eq-photo').innerHTML = `<span class="text-xs text-gray-500">Nenhuma foto</span>`;
                    document.getElementById('preview-serial-photo').innerHTML = `<span class="text-xs text-gray-500">Nenhuma foto</span>`;
                    this.renderGroupsForm();
                }
            }

            showHome() {
                this.currentInspectionId = null;
                this.formData = this.getEmptyForm();
                document.getElementById('eq-model').value = '';
                document.getElementById('eq-serial').value = '';
                document.getElementById('eq-meter').value = '';
                document.getElementById('eq-location').value = '';
                document.getElementById('eq-inspector').value = '';
                document.getElementById('eq-date').value = new Date().toISOString().split('T')[0];
                document.getElementById('preview-eq-photo').innerHTML = `<span class="text-xs text-gray-500">Nenhuma foto</span>`;
                document.getElementById('preview-serial-photo').innerHTML = `<span class="text-xs text-gray-500">Nenhuma foto</span>`;
                this.renderGroupsForm();

                document.getElementById('view-form').classList.remove('hidden');
                document.getElementById('view-history').classList.add('hidden');
                document.getElementById('view-report').classList.add('hidden');
            }

            showHistory() {
                const list = document.getElementById('history-list');
                list.innerHTML = '';

                if (this.inspections.length === 0) {
                    list.innerHTML = `<div class="bg-xcmgDark border border-gray-800 p-8 rounded-xl text-center text-gray-400">Nenhum relatório salvo no dispositivo.</div>`;
                } else {
                    this.inspections.forEach(item => {
                        const div = document.createElement('div');
                        div.className = 'bg-xcmgDark border border-gray-800 p-4 rounded-xl flex flex-col sm:flex-row justify-between items-start sm:items-center gap-3';
                        div.innerHTML = `
                            <div>
                                <h3 class="font-bold text-white text-base">${item.model || 'Equipamento XCMG'} <span class="text-xs text-gray-400 font-normal">(${item.serial || 'S/N não inf.'})</span></h3>
                                <p class="text-xs text-gray-400">Data: ${item.date} &bull; Horímetro: ${item.meter || 'N/I'} &bull; Inspetor: ${item.inspector || 'Não inf.'}</p>
                            </div>
                            <div class="flex space-x-2 w-full sm:w-auto">
                                <button onclick="app.viewReport('${item.id}')" class="flex-1 sm:flex-none bg-gray-800 hover:bg-gray-700 text-white px-3 py-1.5 rounded-lg text-xs font-semibold transition">
                                    <i class="fa-solid fa-eye mr-1"></i> Ver
                                </button>
                                <button onclick="app.editReport('${item.id}')" class="flex-1 sm:flex-none bg-xcmgYellow hover:bg-yellow-500 text-gray-900 px-3 py-1.5 rounded-lg text-xs font-bold transition">
                                    <i class="fa-solid fa-pen-to-square mr-1"></i> Editar
                                </button>
                                <button onclick="app.deleteReport('${item.id}')" class="bg-red-900/40 hover:bg-red-900 text-red-300 px-3 py-1.5 rounded-lg text-xs font-semibold transition">
                                    <i class="fa-solid fa-trash"></i>
                                </button>
                            </div>
                        `;
                        list.appendChild(div);
                    });
                }

                document.getElementById('view-form').classList.add('hidden');
                document.getElementById('view-history').classList.remove('hidden');
                document.getElementById('view-report').classList.add('hidden');
            }

            viewReport(id) {
                const item = this.inspections.find(i => i.id === id);
                if (!item) return;
                this.currentInspectionId = id;

                document.getElementById('rep-title').innerText = `Relatório: ${item.model} (${item.serial || 'S/N'})`;
                document.getElementById('rep-subtitle').innerText = `Realizado em ${item.date} por ${item.inspector || 'Inspetor'} - Local: ${item.location || 'N/I'}`;

                const content = document.getElementById('report-content');
                let html = `
                    <div class="grid grid-cols-2 sm:grid-cols-4 gap-4 bg-gray-900 p-4 rounded-lg border border-gray-800 text-xs">
                        <div><span class="text-gray-400 block">Horímetro:</span> <strong class="text-white">${item.meter || 'N/I'}</strong></div>
                        <div><span class="text-gray-400 block">Local:</span> <strong class="text-white">${item.location || 'N/I'}</strong></div>
                        <div><span class="text-gray-400 block">Data:</span> <strong class="text-white">${item.date}</strong></div>
                        <div><span class="text-gray-400 block">Inspetor:</span> <strong class="text-white">${item.inspector || 'N/I'}</strong></div>
                    </div>
                `;

                if (item.eqPhoto || item.serialPhoto) {
                    html += `<div class="grid grid-cols-2 gap-4">`;
                    if (item.eqPhoto) html += `<div><span class="text-xs text-gray-400 block mb-1">Foto Equipamento</span><img src="${item.eqPhoto}" class="rounded-lg h-40 w-full object-cover border border-gray-800"></div>`;
                    if (item.serialPhoto) html += `<div><span class="text-xs text-gray-400 block mb-1">Foto Número de Série</span><img src="${item.serialPhoto}" class="rounded-lg h-40 w-full object-cover border border-gray-800"></div>`;
                    html += `</div>`;
                }

                GROUPS_DEF.forEach(gName => {
                    const g = item.groups[gName];
                    const badgeColor = g.status === 'OK' ? 'bg-green-600 text-white' : (g.status === 'Atencao' ? 'bg-yellow-600 text-white' : 'bg-red-600 text-white');
                    
                    html += `
                        <div class="border border-gray-800 rounded-lg p-4 space-y-3 bg-gray-900">
                            <div class="flex justify-between items-center">
                                <h4 class="font-bold text-white text-sm">${gName}</h4>
                                <span class="px-2.5 py-0.5 rounded text-xs font-bold ${badgeColor}">${g.status}</span>
                            </div>
                            ${g.obs ? `<p class="text-xs text-gray-300"><strong>Obs:</strong> ${g.obs}</p>` : ''}
                            ${g.photo ? `<div class="mt-2"><img src="${g.photo}" class="rounded-lg h-32 w-full object-cover border border-gray-800"></div>` : ''}
                            
                            ${g.parts && g.parts.length > 0 ? `
                                <div class="mt-2 border-t border-gray-800 pt-2">
                                    <span class="text-xs font-semibold text-xcmgYellow block mb-1">Peças Solicitadas:</span>
                                    <div class="space-y-1">
                                        ${g.parts.map(p => `
                                            <div class="flex justify-between text-xs bg-gray-800 p-1.5 rounded">
                                                <span>${p.desc || 'Peça'} (PN: <strong class="text-white">${p.pn || 'N/I'}</strong>)</span>
                                                <span class="font-bold text-xcmgYellow">Qtd: ${p.qty || 1}</span>
                                            </div>
                                        `).join('')}
                                    </div>
                                </div>
                            ` : ''}
                        </div>
                    `;
                });

                content.innerHTML = html;

                document.getElementById('view-form').classList.add('hidden');
                document.getElementById('view-history').classList.add('hidden');
                document.getElementById('view-report').classList.remove('hidden');
            }

            editReport(id) {
                const item = this.inspections.find(i => i.id === id);
                if (!item) return;
                
                this.formData = JSON.parse(JSON.stringify(item));
                
                document.getElementById('eq-model').value = this.formData.model || '';
                document.getElementById('eq-serial').value = this.formData.serial || '';
                document.getElementById('eq-meter').value = this.formData.meter || '';
                document.getElementById('eq-location').value = this.formData.location || '';
                document.getElementById('eq-inspector').value = this.formData.inspector || '';
                document.getElementById('eq-date').value = this.formData.date || '';

                if (this.formData.eqPhoto) {
                    document.getElementById('preview-eq-photo').innerHTML = `<img src="${this.formData.eqPhoto}" class="w-full h-full object-cover">`;
                } else {
                    document.getElementById('preview-eq-photo').innerHTML = `<span class="text-xs text-gray-500">Nenhuma foto</span>`;
                }

                if (this.formData.serialPhoto) {
                    document.getElementById('preview-serial-photo').innerHTML = `<img src="${this.formData.serialPhoto}" class="w-full h-full object-cover">`;
                } else {
                    document.getElementById('preview-serial-photo').innerHTML = `<span class="text-xs text-gray-500">Nenhuma foto</span>`;
                }

                this.renderGroupsForm();

                document.getElementById('view-form').classList.remove('hidden');
                document.getElementById('view-history').classList.add('hidden');
                document.getElementById('view-report').classList.add('hidden');
            }

            deleteReport(id) {
                if (confirm('Tem certeza que deseja excluir este relatório?')) {
                    this.inspections = this.inspections.filter(i => i.id !== id);
                    localStorage.setItem('xcmg_inspections', JSON.stringify(this.inspections));
                    this.showHistory();
                }
            }

            generatePDFFromCurrent() {
                const item = this.inspections.find(i => i.id === this.currentInspectionId);
                if (!item) return;

                const { jsPDF } = window.jspdf;
                const doc = new jsPDF('p', 'mm', 'a4');
                let y = 15;

                // Cabeçalho
                doc.setFillColor(30, 34, 41);
                doc.rect(0, 0, 210, 25, 'F');
                doc.setTextColor(255, 184, 0);
                doc.setFontSize(16);
                doc.setFont('helvetica', 'bold');
                doc.text('XCMG - FLEETCHECK PRO', 14, 16);
                doc.setTextColor(255, 255, 255);
                doc.setFontSize(10);
                doc.text('Relatório Técnico de Inspeção', 150, 16);

                y = 35;
                doc.setTextColor(40, 40, 40);
                doc.setFontSize(12);
                doc.setFont('helvetica', 'bold');
                doc.text(`Equipamento: ${item.model || 'N/I'}`, 14, y);
                y += 7;

                doc.setFontSize(10);
                doc.setFont('helvetica', 'normal');
                doc.text(`Série / Chassi: ${item.serial || 'N/I'}`, 14, y);
                doc.text(`Horímetro: ${item.meter || 'N/I'}`, 110, y);
                y += 6;
                doc.text(`Local / Obra: ${item.location || 'N/I'}`, 14, y);
                doc.text(`Data: ${item.date}`, 110, y);
                y += 6;
                doc.text(`Inspetor: ${item.inspector || 'N/I'}`, 14, y);
                y += 10;

                // Linha divisória
                doc.setDrawColor(200, 200, 200);
                doc.line(14, y, 196, y);
                y += 8;

                // Grupos
                GROUPS_DEF.forEach((gName, index) => {
                    const g = item.groups[gName];
                    if (y > 260) {
                        doc.addPage();
                        y = 20;
                    }

                    doc.setFont('helvetica', 'bold');
                    doc.setFontSize(11);
                    doc.text(`${index + 1}. ${gName}`, 14, y);
                    
                    // Status Badge Cor
                    doc.setFontSize(10);
                    if (g.status === 'OK') doc.setTextColor(0, 128, 0);
                    else if (g.status === 'Atencao') doc.setTextColor(200, 150, 0);
                    else doc.setTextColor(200, 0, 0);
                    doc.text(`Status: ${g.status}`, 160, y);
                    doc.setTextColor(40, 40, 40);
                    y += 6;

                    if (g.obs) {
                        doc.setFont('helvetica', 'normal');
                        doc.setFontSize(9);
                        const splitObs = doc.splitTextToSize(`Obs: ${g.obs}`, 180);
                        doc.text(splitObs, 14, y);
                        y += (splitObs.length * 5) + 3;
                    }

                    if (g.parts && g.parts.length > 0) {
                        doc.setFont('helvetica', 'bold');
                        doc.setFontSize(9);
                        doc.text('Peças Solicitadas:', 14, y);
                        y += 5;
                        g.parts.forEach(p => {
                            doc.setFont('helvetica', 'normal');
                            doc.text(`- ${p.desc || 'Peça'} | PN: ${p.pn || 'N/I'} | Qtd: ${p.qty || 1}`, 18, y);
                            y += 5;
                        });
                        y += 3;
                    }

                    y += 4;
                });

                // Salvar PDF
                doc.save(`Relatorio_Inspecao_${item.model || 'XCMG'}_${item.date}.pdf`);
            }
        }
